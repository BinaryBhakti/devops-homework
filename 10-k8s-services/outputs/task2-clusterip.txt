# three ways to reach the same Service from inside the cluster
$ kubectl exec curl-client -- curl -s http://web-service-clusterip:8080 | grep -i '<title>'
<title>Welcome to nginx!</title>

$ kubectl exec curl-client -- curl -s http://web-service-clusterip.default.svc.cluster.local:8080 | grep -i '<title>'

# the VIP load-balances: 20 requests, count how many distinct pods answered
$ kubectl exec curl-client -- sh -c 'for i in $(seq 1 20); do curl -s -o /dev/null -w "%{remote_ip}\n" http://web-service-clusterip:8080; done' | sort | uniq -c
  20 10.100.66.42

########## TASK 2 addendum — actually proving the VIP load-balances ##########

# curl's %{remote_ip} reports the address curl dialled (the VIP), not the backend that answered.
# give each pod a distinguishable page instead, then count the answers.
