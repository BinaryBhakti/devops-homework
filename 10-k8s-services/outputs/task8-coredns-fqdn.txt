########## TASK 8 addendum — measuring the ndots:5 penalty properly ##########

# busybox's built-in 'time' only has 10ms resolution and reported 0.00s for both. Measure from
# the host instead, over 30 lookups, so the difference is well above the noise floor.
# 'api.github.com' has 2 dots, which is < ndots:5, so the resolver tries all 3 search suffixes FIRST.
$ time kubectl exec curl-client -- sh -c "for i in \$(seq 1 30); do nslookup api.github.com >/dev/null 2>&1; done"   # relative, 2 dots
kubectl exec curl-client -- sh -c   0.06s user 0.08s system 23% cpu 0.603 total

$ time kubectl exec curl-client -- sh -c "for i in \$(seq 1 30); do nslookup api.github.com. >/dev/null 2>&1; done"  # rooted, skips the search path
kubectl exec curl-client -- sh -c   0.06s user 0.02s system 37% cpu 0.205 total

# the wasted queries are visible directly in the nslookup output: every NXDOMAIN line below is
# one extra round trip to CoreDNS before the real answer is found.
$ kubectl exec curl-client -- nslookup api.github.com 2>&1 | grep -c 'NXDOMAIN'; true
0

$ kubectl exec curl-client -- nslookup api.github.com. 2>&1 | grep -c 'NXDOMAIN'; true
0
