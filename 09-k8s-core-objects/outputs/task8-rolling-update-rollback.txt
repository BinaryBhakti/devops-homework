$ sed -n '1,25p' 01-rolling-update/deployment-v1.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: app-rolling
  labels:
    app: app-rolling
spec:
  replicas: 4
  # RollingUpdate: Kubernetes replaces old pods one-by-one while keeping the service up.
  # maxSurge: 1     -> Allows 1 extra pod above the desired count during the update.
  # maxUnavailable: 0 -> Zero pods can be unavailable at any point. Zero downtime guaranteed.
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1
      maxUnavailable: 0
  selector:
    matchLabels:
      app: app-rolling
  template:
    metadata:
      labels:
        app: app-rolling
        version: v1
    spec:
