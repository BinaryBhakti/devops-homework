# pre-pull busybox on both nodes first, so the Running phase is not hidden behind the image pull
$ kubectl run warm-busybox --image=busybox:latest --restart=Never --command -- true; kubectl wait --for=jsonpath='{.status.phase}'=Succeeded pod/warm-busybox --timeout=120s; kubectl delete pod warm-busybox
