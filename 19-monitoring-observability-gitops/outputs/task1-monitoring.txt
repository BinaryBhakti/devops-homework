# request rate, error ratio and p95 latency for /api/orders (PromQL)
$ PQ 'sum by (status) (rate(http_requests_total{route="/api/orders"}[1m]))'
   {'status': '200'} => 1.1146
   {'status': '201'} => 0.5308

$ PQ 'histogram_quantile(0.95, sum by (le) (rate(http_request_duration_seconds_bucket{route="/api/orders"}[1m])))'
    => 0.0997

# CPU UTILISATION — the app process (its own process_* metrics) and the whole Docker VM (node-exporter)
$ PQ 'rate(process_cpu_seconds_total{job="orders-api"}[1m])'
   {'service': 'orders-api'} => 0.0514

$ PQ '100 * (1 - avg(rate(node_cpu_seconds_total{mode="idle"}[1m])))'
    => 86.8612

# MEMORY UTILISATION
$ PQ 'process_resident_memory_bytes{job="orders-api"} / 1024 / 1024'
   {'service': 'orders-api'} => 42.2852

$ PQ '100 * (1 - node_memory_MemAvailable_bytes / node_memory_MemTotal_bytes)'
    => 53.1841
