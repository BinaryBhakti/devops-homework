# ROOT CAUSE: k8s.io/minikube-hostpath creates a hostPath PV with NO nodeAffinity. Kubernetes therefore thinks the volume is usable from any node; each node
# silently creates its own empty /tmp/hostpath-provisioner/... directory. ReadWriteOnce is not enforced either — RWO means "one NODE", and nothing tells the
# scheduler which node. The course test passes only if the replacement pod happens to land on the same node.
# FIX — a provisioner that records the volume topology: the CSI hostpath driver addon

# Same claim, but from the CSI hostpath driver's StorageClass. Unlike k8s.io/minikube-hostpath,
# this provisioner records WHICH node holds the volume (PV nodeAffinity), so the scheduler
# places every pod that mounts it on that node.
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: web-data-csi
  namespace: production-webapp
spec:
  accessModes: [ReadWriteOnce]
  storageClassName: csi-hostpath-sc
  resources:
    requests: { storage: 500Mi }
